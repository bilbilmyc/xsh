project(xsh_rdp_native VERSION 1.0.0 LANGUAGES C CXX)
option(XSH_RDP_TEST_PEER "Compile the loopback peer used only by native integration tests" OFF)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WITH_CLIENT_COMMON ON CACHE BOOL "" FORCE)
set(WITH_CLIENT OFF CACHE BOOL "" FORCE)
set(WITH_SERVER ${XSH_RDP_TEST_PEER} CACHE BOOL "" FORCE)
set(WITH_SERVER_CHANNELS ${XSH_RDP_TEST_PEER} CACHE BOOL "" FORCE)
set(WITH_X11 OFF CACHE BOOL "" FORCE)
set(WITH_WAYLAND OFF CACHE BOOL "" FORCE)
set(WITH_SDL OFF CACHE BOOL "" FORCE)
set(WITH_SAMPLE OFF CACHE BOOL "" FORCE)
set(WITH_PROXY OFF CACHE BOOL "" FORCE)
set(WITH_PROXY_APP OFF CACHE BOOL "" FORCE)
set(WITH_SHADOW OFF CACHE BOOL "" FORCE)
set(WITH_PLATFORM_SERVER OFF CACHE BOOL "" FORCE)
set(WITH_WINPR_TOOLS OFF CACHE BOOL "" FORCE)
set(WITH_WINPR_TOOLS_CLI OFF CACHE BOOL "" FORCE)
set(WITH_SERVER_SHADOW_CLI OFF CACHE BOOL "" FORCE)
set(WITH_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(WITH_UNICODE_BUILTIN ON CACHE BOOL "" FORCE)
set(WITH_SMARTCARD_EMULATE OFF CACHE BOOL "" FORCE)
set(WITH_PCSC OFF CACHE BOOL "" FORCE)
set(WITH_PKCS11 OFF CACHE BOOL "" FORCE)
set(WITH_KRB5 OFF CACHE BOOL "" FORCE)
# XSH uses the built-in NLA credentials flow. Do not let WinPR discover JSON
# libraries from the build machine and add an unbundled Homebrew dependency.
set(WITH_JSON_DISABLED ON CACHE BOOL "" FORCE)
set(WITH_AAD OFF CACHE BOOL "" FORCE)
set(WITH_FFMPEG OFF CACHE BOOL "" FORCE)
set(WITH_SWSCALE OFF CACHE BOOL "" FORCE)
set(WITH_DSP_FFMPEG OFF CACHE BOOL "" FORCE)
set(WITH_VIDEO_FFMPEG OFF CACHE BOOL "" FORCE)
set(WITH_OPENH264 OFF CACHE BOOL "" FORCE)
set(WITH_MEDIA_FOUNDATION OFF CACHE BOOL "" FORCE)
set(WITH_VERBOSE_WINPR_ASSERT OFF CACHE BOOL "" FORCE)
set(WITH_CAIRO OFF CACHE BOOL "" FORCE)
set(WITH_CUPS OFF CACHE BOOL "" FORCE)
set(WITH_ALSA OFF CACHE BOOL "" FORCE)
set(WITH_PULSE OFF CACHE BOOL "" FORCE)
set(WITH_FAAC OFF CACHE BOOL "" FORCE)
set(WITH_FAAD2 OFF CACHE BOOL "" FORCE)
set(WITH_FDK_AAC OFF CACHE BOOL "" FORCE)
set(WITH_GSM OFF CACHE BOOL "" FORCE)
set(WITH_OPUS OFF CACHE BOOL "" FORCE)
set(WITH_LAME OFF CACHE BOOL "" FORCE)
set(WITH_SOXR OFF CACHE BOOL "" FORCE)

set(WITH_INTERNAL_RC4 ON CACHE BOOL "" FORCE)
set(WINPR_UTILS_IMAGE_JPEG OFF CACHE BOOL "" FORCE)
set(WINPR_UTILS_IMAGE_PNG OFF CACHE BOOL "" FORCE)
set(WINPR_UTILS_IMAGE_WEBP OFF CACHE BOOL "" FORCE)
set(CHANNEL_RDP2TCP OFF CACHE BOOL "" FORCE)
set(CHANNEL_RDPECAM OFF CACHE BOOL "" FORCE)
set(CHANNEL_SSHAGENT OFF CACHE BOOL "" FORCE)
set(CHANNEL_PRINTER OFF CACHE BOOL "" FORCE)
set(CHANNEL_URBDRC OFF CACHE BOOL "" FORCE)
set(CHANNEL_RDPEWA OFF CACHE BOOL "" FORCE)
set(CHANNEL_RDPEAR OFF CACHE BOOL "" FORCE)
set(WITH_POSITION_INDEPENDENT_CODE ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WITH_CHANNELS ON CACHE BOOL "" FORCE)
set(WITH_CLIENT_CHANNELS ON CACHE BOOL "" FORCE)
file(GLOB channel_dirs "${FREERDP_SOURCE}/channels/*")
foreach(channel_dir ${channel_dirs})
  get_filename_component(channel_name "${channel_dir}" NAME)
  if(IS_DIRECTORY "${channel_dir}" AND NOT channel_name STREQUAL "client" AND NOT channel_name STREQUAL "server")
    string(TOUPPER "${channel_name}" channel_option)
    if(channel_name MATCHES "^(cliprdr|drdynvc|disp|rdpgfx)$")
      set("CHANNEL_${channel_option}" ON CACHE BOOL "" FORCE)
      set("CHANNEL_${channel_option}_CLIENT" ON CACHE BOOL "" FORCE)
    else()
      set("CHANNEL_${channel_option}" OFF CACHE BOOL "" FORCE)
      set("CHANNEL_${channel_option}_CLIENT" OFF CACHE BOOL "" FORCE)
    endif()
    if(XSH_RDP_TEST_PEER AND channel_name MATCHES "^(cliprdr|disp)$")
      set("CHANNEL_${channel_option}_SERVER" ON CACHE BOOL "" FORCE)
    else()
      set("CHANNEL_${channel_option}_SERVER" OFF CACHE BOOL "" FORCE)
    endif()
  endif()
endforeach()
add_subdirectory("${FREERDP_SOURCE}" freerdp EXCLUDE_FROM_ALL)
add_library(xsh-rdp-native SHARED xsh_rdp.c)
if(XSH_RDP_TEST_PEER)
  target_sources(xsh-rdp-native PRIVATE test_peer.c)
  target_include_directories(xsh-rdp-native PRIVATE "${OPENSSL_INCLUDE_DIR}")
  target_link_libraries(xsh-rdp-native PRIVATE freerdp-server "${OPENSSL_CRYPTO_LIBRARY}")
  if(MSVC)
    target_compile_options(xsh-rdp-native PRIVATE /Zi)
    target_link_options(xsh-rdp-native PRIVATE /DEBUG)
  endif()
endif()
target_compile_features(xsh-rdp-native PRIVATE c_std_11)
if(MSVC)
  target_compile_options(xsh-rdp-native PRIVATE /we4013 /we4024 /we4047)
endif()
target_compile_definitions(xsh-rdp-native PRIVATE XSH_RDP_EXPORTS FREERDP_EXPORTS FREERDP_CLIENT_EXPORTS WINPR_EXPORTS)
target_include_directories(xsh-rdp-native PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}"
  "${FREERDP_SOURCE}/include"
  "${FREERDP_SOURCE}/winpr/include"
  "${CMAKE_CURRENT_BINARY_DIR}/freerdp/include"
  "${CMAKE_CURRENT_BINARY_DIR}/freerdp/winpr/include")
target_link_libraries(xsh-rdp-native PRIVATE freerdp-client freerdp winpr)
if(APPLE)
  set_target_properties(xsh-rdp-native PROPERTIES INSTALL_NAME_DIR "@rpath")
endif()
install(TARGETS xsh-rdp-native RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
